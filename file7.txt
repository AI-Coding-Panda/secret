// UPDATE

})

db.users.deleteMany({ department: "HR" })
// AGGREGATION
db.users.aggregate([{ $group: { _id: null, totalSales: { $sum:
"$totalSales" } } }])
db.users.aggregate([{ $group: { _id: null, averageAge: { $avg: "$age" } }
}])
db.users.aggregate([{ $group: { _id: "$department", departmentSales: {
$sum: "$totalSales" } } }])
// INDEXES
db.users.createIndex({ email: 1 })
db.users.createIndex({ department: 1 })

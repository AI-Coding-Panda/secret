// FILTER QUERIES
db.posts.find({ likes: { $gt: 100 } })
db.posts.find({ hashtags: "AI" })
db.posts.find({
createdAt: {
$gte: ISODate("2025-01-01T00:00:00Z"),
$lt: ISODate("2025-02-01T00:00:00Z")
}
})
// TOP TRENDING HASHTAGS
db.posts.aggregate([
{ $unwind: "$hashtags" },
{ $group: { _id: "$hashtags", count: { $sum: 1 } } },
{ $sort: { count: -1 } },
{ $limit: 5 }
])
// ADDITIONAL ANALYTICS
